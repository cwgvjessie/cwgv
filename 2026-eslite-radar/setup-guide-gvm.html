專案：職場雷達 × 誠品線上 合作專案  |  負責方：遠見 GVM  |  文件版本：2026-06-11

你的角色：負責兩支內容頁面（gvm-radarlist.html、gvm-article.html）， 這兩支頁面安裝了 GTM 追蹤代碼，是整個合作案的事件追蹤核心。
你需要：正確安裝 GTM、處理來自摩速 iframe 的 postMessage 事件、以及維護所有 dataLayer 事件推送邏輯。

檔案一：gvm-radarlist.html（職場雷達列表頁）

1-1 GTM 代碼段安裝位置

在 <head> 最頂端（其他 script 之前）先推送 dataLayer 初始化，再放 GTM 代碼段：

<!-- 【必須在 GTM 之前】頁面類型標記 -->
<script>
  window.dataLayer = window.dataLayer || [];
  window.dataLayer.push({ page_type: 'radar_list' });
</script>

<!-- Google Tag Manager -->
<script>(function(w,d,s,l,i){w[l]=w[l]||[];w[l].push({'gtm.start':
new Date().getTime(),event:'gtm.js'});var f=d.getElementsByTagName(s)[0],
j=d.createElement(s),dl=l!='dataLayer'?'&l='+l:'';j.async=true;j.src=
'https://www.googletagmanager.com/gtm.js?id='+i+dl;f.parentNode.insertBefore(j,f);
})(window,document,'script','dataLayer','GTM-5ZH7TSVB');</script>
<!-- End Google Tag Manager -->

在 <body> 開頭加 noscript 備援：

<noscript><iframe src="https://www.googletagmanager.com/ns.html?id=GTM-5ZH7TSVB"
height="0" width="0" style="display:none;visibility:hidden"></iframe></noscript>

1-2 事件追蹤腳本

在 </body> 前加入以下追蹤腳本：

<script>
(function () {
  window.dataLayer = window.dataLayer || [];

  // 主列表文章點擊
  document.querySelectorAll('.articles-wrap .article-link').forEach(function (link, idx) {
    link.addEventListener('click', function () {
      window.dataLayer.push({
        event: 'radar_article_click',
        click_area: 'main_list',
        article_position: idx + 1,
        article_category: (link.querySelector('.cat-pill') || {}).textContent || '',
        article_title: (link.querySelector('.article-headline') || {}).textContent || '',
        link_url: link.href
      });
    });
  });

  // 近期熱門文章按鈕點擊
  var recentBtn = document.getElementById('recentBtn');
  if (recentBtn) {
    recentBtn.addEventListener('click', function () {
      window.dataLayer.push({ event: 'radar_recent_open' });
    });
  }

  // Modal 內文章點擊（事件委派）
  var modalBody = document.getElementById('modalBody');
  if (modalBody) {
    modalBody.addEventListener('click', function (e) {
      var row = e.target.closest('.article-row');
      if (!row) return;
      var section = row.closest('.date-section');
      var rows = section ? Array.prototype.slice.call(section.querySelectorAll('.article-row')) : [];
      window.dataLayer.push({
        event: 'radar_article_click',
        click_area: 'recent_modal',
        article_position: rows.indexOf(row) + 1,
        recent_reco_date: section ? (section.querySelector('.date-heading') || {}).textContent || '' : '',
        article_category: (row.querySelector('.cat-pill') || {}).textContent || '',
        article_title: (row.querySelector('.article-row-title') || {}).textContent || '',
        link_url: row.href
      });
    });
  }
})();
</script>

1-3 iframe 高度回傳（給誠品用）

<script>
(function () {
  function sendHeight() {
    var h = Math.max(document.body.scrollHeight, document.documentElement.scrollHeight);
    window.parent.postMessage({ iframeHeight: h }, '*');
  }
  window.addEventListener('load', function () {
    sendHeight();
    setTimeout(sendHeight, 600);
  });
  window.addEventListener('resize', sendHeight);
  new MutationObserver(sendHeight).observe(document.body, { childList: true, subtree: true });
})();
</script>

2-3 postMessage 監聽器（接收摩速事件 + 高度）

這段程式碼讓你的頁面能接收摩速 iframe 傳來的事件，並 push 到 GTM dataLayer：

<script>
(function () {
  window.addEventListener('message', function (e) {
    // 高度自適應
    if (e.data && e.data.iframeHeight) {
      var f = document.getElementById('book-frame');
      if (f) f.style.height = e.data.iframeHeight + 'px';
    }
    // 接收摩速的追蹤事件，轉發到 GTM dataLayer
    if (e.data && e.data.event === 'book_buy_click') {
      window.dataLayer = window.dataLayer || [];
      window.dataLayer.push(Object.assign({ event: e.data.event }, e.data));
    }
  });
})();
</script>
⚠️ 安全性注意：目前監聽器沒有限制 postMessage 來源（origin 為 *）。
如需加強安全性，可改為：if (e.origin !== '摩速的網域') return;

2-4 「繼續閱讀」展開追蹤

3-2 postMessage 監聽器（接收 aigc.html 事件）

本頁直接嵌入摩速的 aigc.html（Layer 3），需監聽其傳出的事件並 push 至 dataLayer：

<script>
(function () {
  window.dataLayer = window.dataLayer || [];

  var ALLOWED_EVENTS = [
    'aigc_back-to-article_click',  // 「回到原始文章」CTA 點擊
    'book_buy_click'               // 誠品選書「立即入手」點擊
  ];

  window.addEventListener('message', function (e) {
    var data = e.data;
    if (!data || typeof data !== 'object') return;

    // 高度自適應：中繼給上層 eslite-aigc.html
    if (typeof data.iframeHeight === 'number') {
      var frame = document.getElementById('aigcFrame');
      if (frame) frame.style.height = data.iframeHeight + 'px';
      try { window.parent.postMessage({ iframeHeight: data.iframeHeight }, '*'); } catch (_) {}
      return;
    }

    // 白名單事件 push 至 GTM dataLayer
    if (data.event && ALLOWED_EVENTS.indexOf(data.event) !== -1) {
      window.dataLayer.push(data);
    }
  });
})();
</script>

所有 GA4 事件一覽

事件名稱觸發時機關鍵參數來源
page_view 頁面載入 page_type、partner_site GTM 自動
radar_article_click 點擊文章列表或 Modal 內文章 click_area、article_position、article_category、article_title、link_url gvm-radarlist.html 腳本
radar_recent_open 點擊「近期熱門文章」按鈕 — gvm-radarlist.html 腳本
article_read_more 點擊「繼續閱讀」展開文章 — gvm-article.html 腳本
book_buy_click 點擊「立即入手」 book_title、link_url 摩速 → postMessage → 你的 listener
aigc_back-to-article_click 點擊「回到原始文章」CTA link_text、link_url 摩速 aigc.html → postMessage → gvm-aigc.html listener

partner_site 說明

partner_site 是 GTM 自訂 JavaScript 變數，讀取 URL 的 ?partner= 參數，用來區分是哪個合作夥伴嵌入了你的頁面。

合作夥伴iframe src 範例GA4 記錄值
誠品gvm-radarlist.html?partner=eslite.comeslite.com
其他未來合作方 Agvm-radarlist.html?partner=partner-a.compartner-a.com

如果誠品沒有傳 ?partner= 參數，GA4 會記錄 (none)。請提醒誠品工程師加上此參數。
